class CenPackedFiles(GeoFlowResource):
    """
    Abstract class for CEN archives to reduce the number of small files.
    """

    _abstract = True,
    _footprint = [
        cendateperiod_deco,
        dict(
            info = 'Any packed file covering a given period',
            attr = dict(
                nativefmt = dict(
                    values = ['tar', 'tar.gz'],
                    default = 'tar'

                    info = "First date of the archive",
                ),

                    optional = True,
                ),

                    optional = True,

                model = dict(
                    optional = True,

        return self.kind
